// p5.js: 인터럽트 방식
// 마우스 위치값 송신, 가변저항값 수신

let port;
let value1 = 0, value2 = 0;

function setup() {
  createCanvas(500, 500);
  port = new p5.SerialPort();
  port.open("COM5");
  port.on('open', serialOpen);
  port.on('data', serialEvent);
}

function draw() {
  background(value1);
  fill(255, 255, 0);
  ellipse(width/2, height/2, value2, value2);
}

function serialOpen() {
  print("Start!");
  port.clear();
  port.write(0);
}

function serialEvent() {
  value2 = port.read();
  value2 = int(map(value2, 0, 255, 0, width));
  value1 = int(map(mouseX, 0, width, 0, 255));
  value1 = constrain(value1, 0, 255);
  port.write(value1);
}
