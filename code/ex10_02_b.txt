// p5.js: 키보드 정수(1∼4) 송신

let port;

function setup() {
  createCanvas(500, 300);
  port = new p5.SerialPort();
  port.open("COM5");

  background(0);
  fill(255, 255, 0);
  textAlign(CENTER);
  textSize(30);
  text("Press on the Number(1,2,3,4)", width/2, height/2);
}

function draw() {
}

function keyPressed() {
  if (key === '1') {
    port.write(1);
  } else if (key === '2') {
    port.write(2);
  } else if (key === '3') {
    port.write(3);
  } else if (key === '4') {
    port.write(4);
  } else {
    port.write(0);
  }
}
