// p5.js: 버튼값 수신하여 원 색상 변경
// if문 사용: 지연 동작

let port;
let value = 0;

function setup() {
  createCanvas(500, 500);
  port = new p5.SerialPort();
  port.open("COM5");
}

function draw() {
  background(0);
  if (port.available() >= 1) {
    value = port.read();
  }
  if (value === 0) {
    fill(255, 255, 0);
  } else {
    fill(255, 0, 0);
  }
  ellipse(width/2, height/2, 200, 200);
}
