// p5.js: 인터럽트 방식
// 가변저항값 수신

let port;
let value = 0;

function setup( ) {
  createCanvas(500, 300);
  port = new p5.SerialPort();
  port.open("COM5");
  port.on('data', serialEvent);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textAlign(CENTER);
  textSize(50);
  text("Variable R: " + nf(value, 3), width/2, height/2);
}

function serialEvent() {
  value = port.read();	
  port.write('B');
}

function mousePressed() {
  print("Start!");
  port.clear();
  port.write('A');
}
