// p5.js: 가변저항값 수신하여 표시 2

let port;
let value = 0;

function setup( ) {
  createCanvas(500, 300);
  port = new p5.SerialPort();
  port.open("COM5");
  port.on('data', serialEvent);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textAlign(CENTER);
  textSize(50);
  text("Variable R: " + nf(value, 4), width/2, height/2);
}

// 수신값이 있으면 자동 호출
function serialEvent() {
  while (port.available() >= 3) {
    let signal = port.readChar();
    if (signal === 'A') {
      let value1 = port.read();
      let value2 = port.read(); 
      value = value1 * 100 + value2;
    }
  }
}
