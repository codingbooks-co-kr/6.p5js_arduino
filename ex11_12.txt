// 텍스트 파일 읽기1

let dataList;

function preload() {			// 외부 파일의 비동기적 로딩 처리
  dataList = loadStrings("data.txt");	// 텍스트 파일을 읽어 각 행을 문자열 배열로 저장
}

function setup() {
  noCanvas();				// 캔버스(=실행화면) 제거
  createP("Data list:");			// 문단형 텍스트 출력
  let data = join(dataList, "<br>");	// 문자열 배열(dataList)을 단일 문자열(data)로 결합
  createP(data);			// 문단형 텍스트(data) 출력
  print(dataList.length);			// 문자열 배열(dataList)의 크기 출력
}
