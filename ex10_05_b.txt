// p5.js: 마우스 클릭값(문자열) 송신

let port;

function setup() {
  createCanvas(500, 300);
  port = new p5.SerialPort();
  port.open("COM5");

  background(0);
  fill(255, 255, 0);
  textAlign(CENTER);
  textSize(30);
  text("Press on the mouse button", width/2, height/2);
}

function draw() {
  if (mouseIsPressed) {
    port.write("on\n");
  } else {
    port.write("off\n");
  }
}
