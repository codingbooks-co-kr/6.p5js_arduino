// p5.js: 인터럽트 방식
// 마우스 x축 위치값(0∼255) 송신

let port; 
let value = 0;

function setup() {
  createCanvas(500, 300);
  port = new p5.SerialPort();
  port.open("COM5");
  port.on('open', serialOpen);
  port.on('data', serialEvent);
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textAlign(CENTER);
  textSize(50);
  text("Brightness: " + nf(value, 3), width/2, height/2);
}

function serialOpen() {
  print("Start!");
  port.clear();
  port.write('A');
}

function serialEvent() {
  let signal = port.read();
  value = int(map(mouseX, 0, width, 0, 255));
  value = constrain(value, 0, 255);
  port.write(value);
}
